\section*{Legend of Variables (in order of appearance)}
\begin{itemize}
    \item $C_{\text{gov}}(t)$: Governance coherence at time $t$
    \item $N$: Total number of participants ("nodes" in the society)
    \item $ZCM_i$: The $i$th individual’s Zero-Conflict Metric score
    \item $\overline{ZCM}$: The average ZCM across all individuals
    \item $V(t)$: Total value created up to time $t$
    \item $f_{\text{care}}(a_i(\tau))$: Function measuring care in action $a_i$ by person $i$ at time $\tau$
    \item $a_i$: Action by person $i$
    \item $\tau$: Time variable for integration
    \item $L_n$: Knowledge level at step $n$
    \item $\varphi$: Golden ratio ($\approx 1.618$)
    \item $r_n$: Recursive learning rate at step $n$
    \item $O_i$: The occupation/art role assigned to person $i$
    \item $\vec{h}_i$: Person $i$’s harmonic identity vector
    \item $\vec{R}_j$: Role $j$’s harmonic requirements
    \item $\arg\max$: Selects the $j$ (role) that matches best
    \item $T_{ij}$: Trade ratio between parties $i$ and $j$
    \item $G_i$, $G_j$: Gifts, goods, or energy exchanged by $i$ and $j$
    \item $H_n$: Home radius or “bloom” at step $n$
    \item $r_0$: Starting radius (“seed” space)
    \item $E_{\text{action}}(t)$: Total harmonic actions at time $t$
    \item $\delta(a_i, \text{harmonic})$: 1 if $a_i$ is harmonic, 0 otherwise
    \item $\text{Memory}(t)$: Collective memory at time $t$
    \item $\text{Event}_i(t)$: Event logged for person $i$ at time $t$
    \item $\text{avatar\_id}$: Unique identifier for each participant
    \item $\vec{h}$: Harmonic state vector
    \item $\text{sigil}$: Symbol representing event or person
    \item $t$: Time of event (generally)
    \item $P_{n+1}$: Perspective after transformation
    \item $P_n$: Previous perspective
    \item $\text{witness}(P_n)$: Seeing through another’s eyes
    \item $f$: Update function (how witnessing changes perspective)
    \item $A_i$: Authority resonance of person $i$
    \item $\alpha$: Attenuation factor (memory echo strength)
    \item $C_{i,n}$: Care delivered by person $i$ at layer $n$ (recursion depth/generation)
    \item $n$: Generation, time, or recursion depth (varies by context)
\end{itemize}
